[2] compilation de Python3.15

cd /tmp
curl -O https://www.python.org/ftp/python/3.15.0/Python-3.15.0rc2.tar.xz
$ tar -xf Python-3.15.0rc2.tar.xz
$ cd Python-3.15.0rc2
$ ./configure --enable-optimizations
$ make -j$(nproc)
$ sudo make altinstall
$ python3.15 --version
